PRAGMA foreign_keys = ON;

-- 1. TABELA: CATEGORIAS

CREATE TABLE IF NOT EXISTS categorias (

    id_categoria INTEGER PRIMARY KEY,

    nome TEXT UNIQUE NOT NULL

);



-- =====================================================

-- 2. TABELA: PRODUTOS

-- =====================================================



CREATE TABLE IF NOT EXISTS produtos (

    id_produto INTEGER PRIMARY KEY,

    id_categoria INTEGER NOT NULL,

    nome TEXT NOT NULL,

    preco_centavos INTEGER NOT NULL CHECK (preco_centavos > 0),

    ativo INTEGER NOT NULL DEFAULT 1 CHECK (ativo IN (0, 1)),



    FOREIGN KEY (id_categoria)

        REFERENCES categorias(id_categoria)

        ON DELETE RESTRICT

);



-- =====================================================

-- 3. TABELA: MESAS

-- =====================================================



CREATE TABLE IF NOT EXISTS mesas (

    id_mesa INTEGER PRIMARY KEY,

    numero INTEGER NOT NULL UNIQUE CHECK (numero > 0),

    capacidade INTEGER NOT NULL CHECK (capacidade > 0)

);



-- =====================================================

-- 4. TABELA: GARCONS

-- =====================================================



CREATE TABLE IF NOT EXISTS garcons (

    id_garcom INTEGER PRIMARY KEY,

    nome TEXT NOT NULL,

    ativo INTEGER NOT NULL DEFAULT 1 CHECK (ativo IN (0, 1))

);



-- =====================================================

-- 5. TABELA: PEDIDOS

-- =====================================================



CREATE TABLE IF NOT EXISTS pedidos (

    id_pedido INTEGER PRIMARY KEY,

    id_mesa INTEGER NOT NULL,

    id_garcom INTEGER NOT NULL,

    data_hora TEXT NOT NULL DEFAULT (

        strftime('%Y-%m-%d %H:%M:%S', 'now', 'localtime')

    ),

    status TEXT NOT NULL DEFAULT 'ABERTO'

        CHECK (status IN ('ABERTO', 'FINALIZADO', 'CANCELADO')),

    observacao TEXT NULL,



    FOREIGN KEY (id_mesa)

        REFERENCES mesas(id_mesa)

        ON DELETE RESTRICT,



    FOREIGN KEY (id_garcom)

        REFERENCES garcons(id_garcom)

        ON DELETE RESTRICT

);



-- =====================================================

-- 6. TABELA: ITENS_PEDIDO

-- =====================================================



CREATE TABLE IF NOT EXISTS itens_pedido (

    id_item INTEGER PRIMARY KEY,

    id_pedido INTEGER NOT NULL,

    id_produto INTEGER NOT NULL,

    quantidade INTEGER NOT NULL CHECK (quantidade > 0),

    preco_unitario_centavos INTEGER NOT NULL

        CHECK (preco_unitario_centavos > 0),

    observacao TEXT NULL,



    FOREIGN KEY (id_pedido)

        REFERENCES pedidos(id_pedido)

        ON DELETE RESTRICT,



    FOREIGN KEY (id_produto)

        REFERENCES produtos(id_produto)

        ON DELETE RESTRICT

);



-- =====================================================

-- 7. TABELA: PAGAMENTOS

-- =====================================================



CREATE TABLE IF NOT EXISTS pagamentos (

    id_pagamento INTEGER PRIMARY KEY,

    id_pedido INTEGER NOT NULL,

    data_hora TEXT NOT NULL DEFAULT (

        strftime('%Y-%m-%d %H:%M:%S', 'now', 'localtime')

    ),

    forma TEXT NOT NULL

        CHECK (forma IN ('PIX', 'DINHEIRO', 'CARTAO')),

    valor_centavos INTEGER NOT NULL CHECK (valor_centavos > 0),



    FOREIGN KEY (id_pedido)

        REFERENCES pedidos(id_pedido)

        ON DELETE RESTRICT

);



-- =====================================================

-- 8. INDICES PARA CONSULTAS E RELACIONAMENTOS

-- =====================================================



CREATE INDEX IF NOT EXISTS idx_produtos_categoria

    ON produtos(id_categoria);



CREATE INDEX IF NOT EXISTS idx_pedidos_mesa

    ON pedidos(id_mesa);



CREATE INDEX IF NOT EXISTS idx_pedidos_garcom

    ON pedidos(id_garcom);



CREATE INDEX IF NOT EXISTS idx_itens_pedido

    ON itens_pedido(id_pedido);



CREATE INDEX IF NOT EXISTS idx_itens_produto

    ON itens_pedido(id_produto);



CREATE INDEX IF NOT EXISTS idx_pagamentos_pedido

    ON pagamentos(id_pedido);



-- =====================================================

-- 9. TRIGGER: VALIDAR FINALIZACAO DO PEDIDO

-- =====================================================



CREATE TRIGGER IF NOT EXISTS validar_finalizacao_pedido

BEFORE UPDATE OF status ON pedidos

FOR EACH ROW

WHEN NEW.status = 'FINALIZADO'

BEGIN



    -- O pedido precisa ter pelo menos um item.

    SELECT CASE

        WHEN NOT EXISTS (

            SELECT 1

            FROM itens_pedido

            WHERE id_pedido = NEW.id_pedido

        )

        THEN RAISE(

            ABORT,

            'Erro: o pedido precisa ter pelo menos um item.'

        )

    END;



    -- Os pagamentos devem cobrir o total do pedido.

    SELECT CASE

        WHEN COALESCE((

            SELECT SUM(quantidade * preco_unitario_centavos)

            FROM itens_pedido

            WHERE id_pedido = NEW.id_pedido

        ), 0) >

        COALESCE((

            SELECT SUM(valor_centavos)

            FROM pagamentos

            WHERE id_pedido = NEW.id_pedido

        ), 0)

        THEN RAISE(

            ABORT,

            'Erro: pagamento insuficiente para finalizar o pedido.'

        )

    END;



END;



-- =====================================================

-- 10. TRIGGER: IMPEDIR PAGAMENTO ACIMA DO TOTAL

-- =====================================================



CREATE TRIGGER IF NOT EXISTS validar_pagamento_insert

BEFORE INSERT ON pagamentos

FOR EACH ROW

WHEN (

    SELECT status

    FROM pedidos

    WHERE id_pedido = NEW.id_pedido

) = 'ABERTO'

BEGIN



    SELECT CASE

        WHEN NEW.valor_centavos + COALESCE((

            SELECT SUM(valor_centavos)

            FROM pagamentos

            WHERE id_pedido = NEW.id_pedido

        ), 0) >

        COALESCE((

            SELECT SUM(quantidade * preco_unitario_centavos)

            FROM itens_pedido

            WHERE id_pedido = NEW.id_pedido

        ), 0)

        THEN RAISE(

            ABORT,

            'Erro: o pagamento ultrapassa o total do pedido.'

        )

    END;



END;



-- =====================================================

-- 11. TRIGGER: IMPEDIR PAGAMENTO EM PEDIDO ENCERRADO

-- =====================================================



CREATE TRIGGER IF NOT EXISTS impedir_pagamento_pedido_encerrado

BEFORE INSERT ON pagamentos

FOR EACH ROW

WHEN (

    SELECT status

    FROM pedidos

    WHERE id_pedido = NEW.id_pedido

) <> 'ABERTO'

BEGIN

    SELECT RAISE(

        ABORT,

        'Erro: somente pedidos abertos podem receber pagamentos.'

    );

END;



-- =====================================================

-- 12. TRIGGER: IMPEDIR ALTERACOES NOS ITENS ENCERRADOS

-- =====================================================



CREATE TRIGGER IF NOT EXISTS impedir_insercao_item_encerrado

BEFORE INSERT ON itens_pedido

FOR EACH ROW

WHEN (

    SELECT status

    FROM pedidos

    WHERE id_pedido = NEW.id_pedido

) <> 'ABERTO'

BEGIN

    SELECT RAISE(

        ABORT,

        'Erro: nao e permitido adicionar itens a pedidos encerrados.'

    );

END;



CREATE TRIGGER IF NOT EXISTS impedir_alteracao_item_encerrado

BEFORE UPDATE ON itens_pedido

FOR EACH ROW

WHEN (

    SELECT status

    FROM pedidos

    WHERE id_pedido = OLD.id_pedido

) <> 'ABERTO'

BEGIN

    SELECT RAISE(

        ABORT,

        'Erro: nao e permitido alterar itens de pedidos encerrados.'

    );

END;



CREATE TRIGGER IF NOT EXISTS impedir_exclusao_item_encerrado

BEFORE DELETE ON itens_pedido

FOR EACH ROW

WHEN (

    SELECT status

    FROM pedidos

    WHERE id_pedido = OLD.id_pedido

) <> 'ABERTO'

BEGIN

    SELECT RAISE(

        ABORT,

        'Erro: nao e permitido excluir itens de pedidos encerrados.'

    );

END;



-- =====================================================

-- 13. VISUALIZAR TOTAL DOS PEDIDOS

-- =====================================================



CREATE VIEW IF NOT EXISTS vw_resumo_pedidos AS

SELECT

    p.id_pedido,

    p.id_mesa,

    p.id_garcom,

    p.data_hora,

    p.status,



    COALESCE((

        SELECT SUM(i.quantidade * i.preco_unitario_centavos)

        FROM itens_pedido i

        WHERE i.id_pedido = p.id_pedido

    ), 0) AS total_centavos,



    COALESCE((

        SELECT SUM(pg.valor_centavos)

        FROM pagamentos pg

        WHERE pg.id_pedido = p.id_pedido

    ), 0) AS pago_centavos,



    COALESCE((

        SELECT SUM(i.quantidade * i.preco_unitario_centavos)

        FROM itens_pedido i

        WHERE i.id_pedido = p.id_pedido

    ), 0) -

    COALESCE((

        SELECT SUM(pg.valor_centavos)

        FROM pagamentos pg

        WHERE pg.id_pedido = p.id_pedido

    ), 0) AS saldo_centavos



FROM pedidos p;
